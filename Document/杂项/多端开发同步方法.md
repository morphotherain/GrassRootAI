在没有环境的地方开始新一轮更新

```bash
# 先获取最新版本
git pull
# 切到主分支
git checkout main
# 拉取主分支
git pull
# 删除上次用于提交的分支
git branch -D dev-day-20250723
# 创建切换到新的提交分支
git checkout -b dev-day-20250000
#
# 进行修改工作
# 添加修改
git add --all
# 提交到分支
git commit -m "Commit1"
# 推送
git push
# 
# 进行修改工作并重复...
#
git add --all
#
git commit -m "CommitN"
#
git push

```

在有环境的地方验证并合并更新

```bash
# 
git fetch
# 临时分支
git checkout -b temp-dev-day-20250000
# 转到提交分支
git checkout dev-day-20250000
# 查看提交列表
git log --oneline
# 转到临时分支
git checkout temp-dev-day-20250000
# 应用第一次提交
git cherry-pick 0594f87
# 如果有问题的话即时修复并添加
git add -u
# 提交修复
git commit -m "fix Commit1"
# 应用第二次提交
git cherry-pick d956b7b
# 如果和刚才的修改有冲突的话先解决冲突
git add  -u
# 完成pick
git cherry-pick --continue
# ...
# 重复pick直到完成所有提交
#
git checkout main
# 把所有提交折叠起来
git merge --squash temp-dev-day-20250000
# 合成一次提交
git commit -m "202500000-commit"
# 删除临时分支
git branch -D temp-dev-day-20250000
# 删除一次性分支
git branch -D dev-day-20250000
# 删除远程
git push --delete origin dev-day-20250000
# 完成的提交！
git push

# 

```
